% Shared TikZ styles for the pipeline diagrams.
%
% Arrow policy, applied without exception in every flowchart: connectors run
% only horizontally or vertically, are joined with rounded corners, and carry
% one small arrowhead at the destination. Branches fan out from, and merge
% into, an explicit coordinate, so a split never produces two arrowheads on
% the same node face and no connector is ever drawn diagonally.
\tikzset{
  block/.style   = {draw, rounded corners=2pt, align=center,
                    font=\footnotesize, inner sep=4pt, minimum height=8mm},
  modelblk/.style= {block, draw=cModel, fill=cModel!8, text=cModel},
  flowblk/.style = {block, draw=cFlow, fill=cFlow!10, text=cFlow!70!black},
  plainblk/.style= {block, draw=cMuted, fill=cLight},
  gate/.style    = {draw=cAccent, fill=cAccent!8, diamond, aspect=2.2,
                    align=center, font=\footnotesize, inner sep=1pt},
  % fl  = a connector that ends in an arrowhead
  % seg = a connector segment that feeds a junction, so carries no head
  fl/.style      = {-{Latex[length=1.7mm,width=1.4mm]}, draw=cMuted,
                    line width=0.65pt, rounded corners=1.4mm},
  seg/.style     = {fl, -},
  lbl/.style     = {font=\scriptsize\itshape, text=cMuted, inner sep=2pt},
  jn/.style      = {circle, fill=cMuted, inner sep=0pt, minimum size=1.5mm},
}
